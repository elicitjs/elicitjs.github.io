// ── This chart's route: a spiral ────────────────────────────────────────
// Each year starts at the top and runs clockwise. Its radius shrinks by six
// bands a turn, so the turns of all six years nest without touching.
const CX = 330, CY = 290, R0 = 230, BAND = 12, PITCH = 6 * BAND;
// Every year's spiral is LENGTH px long, so the years share one scale.
const MAX = 1600000, LENGTH = 1227;
const radius = (i) => R0 - i * BAND;

function routeOf(i) {
  let d = "", run = 0, prev = null;
  for (let deg = 0; run < LENGTH; deg += 3) {
    const t = (deg * Math.PI) / 180, r = radius(i) - (PITCH * deg) / 360;
    const p = [CX + r * Math.sin(t), CY - r * Math.cos(t)];
    if (prev) run += Math.hypot(p[0] - prev[0], p[1] - prev[1]);
    d += `${prev ? "L" : "M"}${p[0].toFixed(1)},${p[1].toFixed(1)}`;
    prev = p;
  }
  return d;
}

const COLOURS = ["#e8a2a2", "#8e9fd3", "#b39b7d", "#f0b02c", "#d6ccb8", "#dc1f45"];

const chart = Elicit({
  width: 580, height: 540,
  margins: { top: 0, right: 0, bottom: 0, left: 0 },
  theme: { background: "#ecdcc6" },
  // Assessed value of household and kitchen furniture owned by Black
  // Georgians, in dollars.
  data: [
    { year: 1875, value: 21186 },
    { year: 1880, value: 498532 },
    { year: 1885, value: 736170 },
    { year: 1890, value: 1173624 },
    { year: 1895, value: 1322694 },
    { year: 1899, value: 1434975 },
  ],
  schema: {
    year:  { type: "ordinal" },
    value: { type: "quantitative", domain: [0, MAX] },
  },
  constraints: [snap({ step: 1 })],
  marks: [
    track({
      path: (d, i) => routeOf(i),
      channels: {
        length: { field: "value", edit: move() },
        stroke: { fn: (d, i) => COLOURS[i] }, strokeWidth: { value: 9 },
      },
    }),
    text({ channels: {
      x: { value: CX - 12 }, y: { fn: (d, i) => CY - radius(i) },
      text: { fn: (d) => `${d.year} — $${d.value.toLocaleString("en-US")}` },
      textAnchor: { value: "end" }, fontSize: { value: 11 }, fill: { value: "#3b2f27" },
    } }),
  ],
});
mount(chart.dom());
