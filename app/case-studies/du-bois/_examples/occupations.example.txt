// ── This chart's route: one lane, then a tight turn back ────────────────
// In data units, X is men along a lane and Y is a level (one per row). A lane
// holds 85,000. The bar turns down and runs back under itself.
const THRESH = 85000, MAX = 170000;
const STEP = 0.375;                 // how far down the turn steps, in levels

// Two linear scales map the route to pixels: x from [0, THRESH], y one row per level.
const X0 = 236, W = 380, TOP = 22, ROW = 32;
const x = (v) => X0 + (v / THRESH) * W;
const y = (level) => TOP + level * ROW;

// The longest bar a row allows: right, then back left one step down.
const routeOf = (i) =>
  `M${x(0)},${y(i)}L${x(THRESH)},${y(i)}` +
  `M${x(THRESH)},${y(i + STEP)}L${x(0)},${y(i + STEP)}`;

const chart = Elicit({
  width: 640, height: 290,
  margins: { top: 0, right: 0, bottom: 0, left: 0 },
  theme: { background: "#ecdcc6" },
  // Occupations of Black men over ten in Georgia, 1890. The last row is the
  // fifteen smaller trades Du Bois bracketed together, summed.
  data: [
    { occupation: "Agricultural laborers",   men: 98400 },
    { occupation: "Farmers and planters",    men: 83012 },
    { occupation: "Laborers",                men: 28723 },
    { occupation: "Steam railway employees", men: 7440 },
    { occupation: "Servants",                men: 7000 },
    { occupation: "Draymen, hackmen",        men: 4390 },
    { occupation: "Carpenters and joiners",  men: 3761 },
    { occupation: "Fifteen other trades",    men: 15897 },
  ],
  schema: {
    occupation: { type: "categorical" },
    men:        { type: "quantitative", domain: [0, MAX] },
  },
  constraints: [snap({ step: 1 })],
  marks: [
    track({
      path: (d, i) => routeOf(i),
      join: "round",
      channels: {
        length: { field: "men", edit: move() },
        stroke: { value: "#d1213f" }, strokeWidth: { value: 7 },
      },
    }),
    text({ channels: {
      x: { value: 14 }, y: { fn: (d, i) => y(i) },
      text: { fn: (d) => d.occupation.toUpperCase() },
      textAnchor: { value: "start" }, fontSize: { value: 10 }, fill: { value: "#3b2f27" },
    } }),
    text({ channels: {
      x: { value: X0 - 10 }, y: { fn: (d, i) => y(i) },
      text: { fn: (d) => d.men.toLocaleString("en-US") },
      textAnchor: { value: "end" }, fontSize: { value: 10 }, fill: { value: "#3b2f27" },
    } }),
  ],
});
mount(chart.dom());
