// ── This chart's track: a bar that wraps at a threshold ─────────────────
// The track is in data units: X is the value, Y is a level (one per row).
// A bar runs right to THRESH, steps down, runs back left to WRAP * THRESH,
// steps down, and so on. The steps hold no value.
const THRESH = 1500, WRAP = 0, MAX = 6000;
const LANE = (1 - WRAP) * THRESH;   // what each lane after the first holds
const STEP = 0.3;                   // how far down a wrap steps, in levels

// Two linear scales map the track to pixels: x from [0, THRESH], y one row per level.
const X0 = 70, W = 470, TOP = 30, ROW = 36;
const x = (v) => X0 + (v / THRESH) * W;
const y = (level) => TOP + level * ROW;

// How many times a value wraps, and so how many levels its row takes.
const wraps = (v) => (v > THRESH ? Math.ceil((v - THRESH) / LANE - 1e-9) : 0);
// The level a row starts on: the rows before it, plus their wraps. It changes
// as you drag, so the rows below a wrapping bar move down.
const levelOf = (i, rows) =>
  rows.slice(0, i).reduce((level, d) => level + 1 + STEP * wraps(d.value), 0);

// The longest bar the row allows, in data units, then mapped to pixels.
function trackOf(i, rows) {
  let level = levelOf(i, rows), end = THRESH, held = THRESH;
  const points = [[0, level], [THRESH, level]];
  while (held < MAX) {
    level += STEP;
    points.push([end, level, true]);            // the step down holds no value
    end = end === THRESH ? WRAP * THRESH : THRESH;
    points.push([end, level]);
    held += LANE;
  }
  return toPixels(points);
}

// Maps each point through x and y. A step down becomes a half circle that
// bulges outward: right at THRESH, left at WRAP * THRESH.
function toPixels(points) {
  const r = (STEP * ROW) / 2;
  return points.flatMap(([v, level, turn]) => {
    if (!turn) return [[x(v), y(level)]];
    const side = v === THRESH ? 1 : -1, top = y(level) - 2 * r;
    return Array.from({ length: 8 }, (_, a) => {
      const t = ((a + 1) / 8) * Math.PI;
      return [x(v) + side * r * Math.sin(t), top + r - r * Math.cos(t), true];
    });
  });
}

// The x axis over [0, THRESH].
const xAxis = guides.custom(() =>
  [0, 250, 500, 750, 1000, 1250, 1500].flatMap((v) => [
    { type: "line", x1: x(v), x2: x(v), y1: TOP - 14, y2: 420, stroke: "#dddddd", strokeWidth: 0.75, background: true },
    { type: "text", x: x(v), y: 432, text: v.toLocaleString("en-US"), fontSize: 10, fill: "#555555", textAnchor: "middle" },
  ]),
);

const chart = Elicit({
  width: 640, height: 450,
  margins: { top: 0, right: 0, bottom: 0, left: 0 },
  data: [
    { name: "A", value: 3000 }, { name: "B", value: 1750 }, { name: "C", value: 500 },
    { name: "D", value: 2300 }, { name: "E", value: 300 },  { name: "F", value: 800 },
    { name: "G", value: 50 },   { name: "H", value: 150 },
  ],
  schema: {
    name:  { type: "categorical" },
    value: { type: "quantitative", domain: [0, MAX] },
  },
  constraints: [snap({ step: 1 })],
  guides: [xAxis],
  marks: [
    ...foldedBar({
      trackOf, field: "value", per: W / THRESH, groove: W, travel: 480,
      stroke: { field: "value", scale: { range: ["teal", "red"] } }, width: 7,
    }),
    text({ channels: {
      x: { value: X0 - 16 }, y: { fn: (d, i, rows) => y(levelOf(i, rows)) },
      text: { field: "name" }, textAnchor: { value: "end" },
      fontSize: { value: 13 }, fill: { value: "#333333" },
    } }),
    text({ channels: {
      x: { value: X0 + W + 16 }, y: { fn: (d, i, rows) => y(levelOf(i, rows)) },
      text: { fn: (d) => d.value.toLocaleString("en-US") }, textAnchor: { value: "start" },
      fontSize: { value: 11 }, fill: { value: "#555555" },
    } }),
  ],
});
mount(chart.dom());

// ── Helpers: a bar laid along a track (the same on every chart of this page) ──
// A track is the bar's centre line as [x, y, turn] points, in pixels. A step
// marked `turn` joins two lanes and holds no value.

// The first `len` pixels of a track, counting lanes only.
function along(track, len) {
  const out = [track[0]];
  let left = len;
  for (let k = 1; k < track.length && left > 1e-6; k++) {
    const [x0, y0] = track[k - 1], [x1, y1, turn] = track[k];
    if (turn) { out.push(track[k]); continue; }
    const step = Math.hypot(x1 - x0, y1 - y0);
    const t = Math.min(1, left / step);
    out.push([x0 + (x1 - x0) * t, y0 + (y1 - y0) * t]);
    left -= step;
  }
  return out;
}

// A `shape` fits its path to a box, so the path states its own centre and half-size.
function pathOf(points) {
  const xs = points.map((q) => q[0]), ys = points.map((q) => q[1]);
  const [x0, x1, y0, y1] = [Math.min(...xs), Math.max(...xs), Math.min(...ys), Math.max(...ys)];
  return {
    d: "M" + points.map((q) => `${q[0].toFixed(1)},${q[1].toFixed(1)}`).join("L"),
    x: (x0 + x1) / 2, y: (y0 + y1) / 2, size: Math.max(x1 - x0, y1 - y0) / 2,
  };
}

// Two shapes over the rows: a faint groove `groove` pixels long, and the bar,
// `per` pixels to a unit of `field`. Both draw from `field`. The `length`
// channel carries the edit: a `slide` that adds to `field` as you drag down,
// its whole domain over `travel` pixels. The groove takes the same slide, so a
// bar at zero can still be grabbed.
function foldedBar({ trackOf, field, per, groove, travel, stroke, width }) {
  const length = { field, edit: slide({ axis: "y", increase: "down", extent: travel }) };
  const line = (lengthOf, opacity) => {
    const at = (d, i, rows) => pathOf(along(trackOf(i, rows), lengthOf(d, i)));
    return shape({
      path: (d, i, rows) => at(d, i, rows).d,
      channels: {
        x: { fn: (d, i, rows) => at(d, i, rows).x },
        y: { fn: (d, i, rows) => at(d, i, rows).y },
        size: { fn: (d, i, rows) => at(d, i, rows).size },
        fill: { value: "none" }, strokeWidth: { value: width }, opacity: { value: opacity },
        stroke, length,
      },
    });
  };
  return [
    line((d, i) => (typeof groove === "function" ? groove(i) : groove), 0.12),
    line((d) => d[field] * per, 1),
  ];
}
