// ── This chart's route: lanes that fold back and forth ──────────────────
// In data units, X is students along a lane and Y is a level (one per row).
// A lane holds 900. Each `M` starts a lane; the turns hold no value.
const THRESH = 900, MAX = 2700;
const STEP = 0.35;                  // how far down a turn steps, in levels

// Two linear scales map the route to pixels: x from [0, THRESH], y one row per level.
const X0 = 190, W = 400, TOP = 28, ROW = 74;
const x = (v) => X0 + (v / THRESH) * W;
const y = (level) => TOP + level * ROW;

// The longest bar a row allows: three lanes, right, left, right.
function routeOf(i) {
  let d = "";
  for (let lane = 0; lane < MAX / THRESH; lane++) {
    const [from, to] = lane % 2 === 0 ? [0, THRESH] : [THRESH, 0];
    const level = i + lane * STEP;
    d += `M${x(from)},${y(level)}L${x(to)},${y(level)}`;
  }
  return d;
}

const chart = Elicit({
  width: 640, height: 470,
  margins: { top: 0, right: 0, bottom: 0, left: 0 },
  theme: { background: "#ecdcc6" },
  // Number of Black students taking each course of study in Georgia schools.
  data: [
    { course: "Business",     students: 12 },
    { course: "Classical",    students: 98 },
    { course: "Professional", students: 152 },
    { course: "Scientific",   students: 161 },
    { course: "Normal",       students: 383 },
    { course: "Industrial",   students: 2252 },
  ],
  schema: {
    course:   { type: "categorical" },
    students: { type: "quantitative", domain: [0, MAX] },
  },
  constraints: [snap({ step: 1 })],
  marks: [
    track({
      path: (d, i) => routeOf(i),
      join: "round",
      channels: {
        length: { field: "students", edit: move() },
        stroke: { value: "#5c4a3d" }, strokeWidth: { value: 9 },
      },
    }),
    text({ channels: {
      x: { value: 20 }, y: { fn: (d, i) => y(i) },
      text: { fn: (d) => d.course.toUpperCase() },
      textAnchor: { value: "start" }, fontSize: { value: 12 }, fill: { value: "#3b2f27" },
    } }),
    text({ channels: {
      x: { value: X0 - 14 }, y: { fn: (d, i) => y(i) },
      text: { field: "students" },
      textAnchor: { value: "end" }, fontSize: { value: 12 }, fill: { value: "#3b2f27" },
    } }),
  ],
});
mount(chart.dom());
