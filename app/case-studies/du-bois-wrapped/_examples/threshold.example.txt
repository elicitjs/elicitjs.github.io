// ── This chart's route: a bar that wraps at a threshold ─────────────────
// In data units, X is the value and Y is a level (one per row). A bar runs
// right to THRESH, steps down, runs back left to WRAP * THRESH, steps down,
// and so on. Each `M` starts a lane; the steps between lanes hold no value.
const THRESH = 1500, WRAP = 0, MAX = 6000;
const LANE = (1 - WRAP) * THRESH;   // what each lane after the first holds
const STEP = 0.3;                   // how far down a wrap steps, in levels

// Two linear scales map the route to pixels: x from [0, THRESH], y one row per level.
const X0 = 70, W = 470, TOP = 30, ROW = 36;
const x = (v) => X0 + (v / THRESH) * W;
const y = (level) => TOP + level * ROW;

// How many times a value wraps, and so how many levels its row takes.
const wraps = (v) => (v > THRESH ? Math.ceil((v - THRESH) / LANE - 1e-9) : 0);
// The level a row starts on: the rows before it, plus their wraps. It changes
// as you drag, so the rows below a wrapping bar move down.
const levelOf = (i, rows) =>
  rows.slice(0, i).reduce((level, d) => level + 1 + STEP * wraps(d.value), 0);

// The longest bar a row allows, as one lane per `M`. It holds MAX.
function routeOf(i, rows) {
  let level = levelOf(i, rows), from = 0, to = THRESH, held = THRESH;
  let d = `M${x(from)},${y(level)}L${x(to)},${y(level)}`;
  while (held < MAX) {
    level += STEP;
    [from, to] = [to, to === THRESH ? WRAP * THRESH : THRESH];
    d += `M${x(from)},${y(level)}L${x(to)},${y(level)}`;
    held += LANE;
  }
  return d;
}

// The x axis over [0, THRESH].
const xAxis = guides.custom(() =>
  [0, 250, 500, 750, 1000, 1250, 1500].flatMap((v) => [
    { type: "line", x1: x(v), x2: x(v), y1: TOP - 14, y2: 420, stroke: "#dddddd", strokeWidth: 0.75, background: true },
    { type: "text", x: x(v), y: 432, text: v.toLocaleString("en-US"), fontSize: 10, fill: "#555555", textAnchor: "middle" },
  ]),
);

const chart = Elicit({
  width: 640, height: 450,
  margins: { top: 0, right: 0, bottom: 0, left: 0 },
  data: [
    { name: "A", value: 3000 }, { name: "B", value: 1750 }, { name: "C", value: 500 },
    { name: "D", value: 2300 }, { name: "E", value: 300 },  { name: "F", value: 800 },
    { name: "G", value: 50 },   { name: "H", value: 150 },
  ],
  schema: {
    name:  { type: "categorical" },
    value: { type: "quantitative", domain: [0, MAX] },
  },
  constraints: [snap({ step: 1 })],
  guides: [xAxis],
  marks: [
    track({
      path: (d, i, rows) => routeOf(i, rows),
      join: "round",
      channels: {
        length: { field: "value", edit: move() },
        stroke: { field: "value", scale: { range: ["teal", "red"] } },
        strokeWidth: { value: 7 },
      },
    }),
    text({ channels: {
      x: { value: X0 - 16 }, y: { fn: (d, i, rows) => y(levelOf(i, rows)) },
      text: { field: "name" }, textAnchor: { value: "end" },
      fontSize: { value: 13 }, fill: { value: "#333333" },
    } }),
    text({ channels: {
      x: { value: X0 + W + 16 }, y: { fn: (d, i, rows) => y(levelOf(i, rows)) },
      text: { fn: (d) => d.value.toLocaleString("en-US") }, textAnchor: { value: "start" },
      fontSize: { value: 11 }, fill: { value: "#555555" },
    } }),
  ],
});
mount(chart.dom());
