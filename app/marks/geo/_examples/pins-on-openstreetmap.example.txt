const chart = Elicit({
  width: 520, height: 360,
  margins: { top: 0, right: 0, bottom: 0, left: 0 },
  // No GeoJSON needed: fit the projection to a two-corner lon/lat box.
  projection: {
    type: "mercator",
    domain: {
      type: "MultiPoint",
      coordinates: [
        [-123.225, 49.198], // [west, south]
        [-123.023, 49.297], // [east, north]
      ],
    },
  },
  data: [
    { lon: -123.1207, lat: 49.2827, place: "Downtown" },
    { lon: -123.1560, lat: 49.2680, place: "Kitsilano" },
  ],
  schema: {
    lon:   { type: "quantitative" },
    lat:   { type: "quantitative" },
    place: { type: "categorical" },
  },
  marks: [
    geoTile(),   // defaults to OSM standard + its attribution

    geoPoint({
      size: 7, fill: "#EE7B5E", stroke: "#fff", strokeWidth: 2,
      channels: {
        lon: { field: "lon" },
        lat: { field: "lat" },
      },
      edits: [edit.geo.move(), edit.geo.create({ defaults: { place: "new pin" } })],
    }),
  ],
});
mount(chart.dom());
