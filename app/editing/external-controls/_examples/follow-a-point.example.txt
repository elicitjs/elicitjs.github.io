// A label of your own that follows a point: read the chart, never reach into it.
const chart = Elicit({
  width: 420, height: 280,
  margins: { top: 20, right: 20, bottom: 40, left: 44 },
  schema: {
    x: { type: "quantitative", domain: [0, 100] },
    y: { type: "quantitative", domain: [0, 100] },
  },
  data: [{ x: 30, y: 60 }, { x: 68, y: 35 }],
  marks: [
    point({
      id: "dot",
      size: 9,
      channels: { x: { field: "x", edit: move({ channels: ["x", "y"] }) }, y: { field: "y" } },
    }),
  ],
});
mount(chart.dom());

const labels = chart.getData().map(() => {
  const el = document.createElement("div");
  el.style.cssText = "position:absolute;left:0;top:0;pointer-events:none;font:12px system-ui;"
    + "padding:2px 6px;border-radius:4px;background:#14283A;color:#fff;white-space:nowrap";
  return el;
});

// After every frame: where is each dot now, and what does its row hold?
chart.on("render", () => {
  const host = chart.dom().getBoundingClientRect();
  for (const view of chart.scene({ mark: "dot" })) {
    const el = labels[view.index];
    if (!el.isConnected) chart.dom().appendChild(el);
    // Plot pixels -> the page -> the chart's own container.
    const at = chart.toClient({ x: view.center.x, y: view.box.y });
    el.style.transform = `translate(${at.x - host.left}px, ${at.y - host.top}px) translate(-50%, -130%)`;
    el.textContent = `${Math.round(view.datum.x)}, ${Math.round(view.datum.y)}`;
  }
});
