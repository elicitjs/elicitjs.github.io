function square(options = {}) {
  const opts = authoring.normalizeMarkOptions(options, { mark: "square", allow: [] });
  const { channels = {} } = opts;
  return {
    // id / edits / constraints / table, passed through verbatim.
    ...authoring.markCommon(opts),
    type: "square",
    channels,
    discreteScale: "point",
    xKey: channels.x && channels.x.field,
    yKey: channels.y && channels.y.field,
    build(data, scales, width, height) {
      return data.map((d, i) => {
        const style = authoring.resolveStyle(scales, channels, d, { fill: "#4f46e5" });
        const cx = authoring.encodeChannel(scales, channels, "x", d, width / 2);
        const cy = authoring.encodeChannel(scales, channels, "y", d, height / 2);
        const r = authoring.encodeChannel(scales, channels, "size", d, 8);
        const side = Math.max(0, r * 2);
        return {
          type: "rect",
          x: cx - side / 2, y: cy - side / 2,
          width: side, height: side,
          ...style, data: d, index: i,
        };
      });
    },
  };
}
